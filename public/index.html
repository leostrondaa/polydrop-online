const ROUND_DROP_LABELS = [ ['Cloreto', 'vinila'], ['Eteno'], ['Estireno'], ['TFE'], ['Propeno'] ]; const MINI_DROP_LABELS = [ 'CV', 'Et', 'Es', 'TFE', 'Pr' ]; function drawDropInLane(r, d, latency) { const yFraction = d.y + (d.vy || 0) * latency; if (yFraction < -.15 || yFraction > 1.18) return; const info = DATA[d.idx] || DATA[0]; const mini = !!r.mini; // Bolinhas redondas em todas as resoluções. const radius = mini ? Math.max(9, Math.min(16, Math.min(r.w, r.h) * .075)) : Math.max(21, Math.min(31, r.w * .084, r.h * .056)); const cy = r.y + Math.max(0, yFraction) * r.h; const cx = r.x + Math.max( radius + 3, Math.min(r.w - radius - 3, d.x * r.w) ); ctx.save(); rr(r.x + 2, r.y + 2, r.w - 4, r.h - 4, mini ? 7 : 11); ctx.clip(); // Brilho colorido da molécula. ctx.shadowColor = info.color; ctx.shadowBlur = mini ? 5 : 12; const glow = ctx.createRadialGradient( cx - radius * .3, cy - radius * .35, 1, cx, cy, radius ); glow.addColorStop(0, '#315465'); glow.addColorStop(.65, '#17384a'); glow.addColorStop(1, '#0b2234'); ctx.fillStyle = glow; ctx.strokeStyle = info.color; ctx.lineWidth = mini ? 1.6 : 2.6; // CÍRCULO, não cápsula. ctx.beginPath(); ctx.arc(cx, cy, radius, 0, Math.PI * 2); ctx.fill(); ctx.stroke(); ctx.shadowBlur = 0; // Pequeno reflexo. ctx.fillStyle = '#ffffff29'; ctx.beginPath(); ctx.arc( cx - radius * .34, cy - radius * .37, mini ? 1.6 : 3, 0, Math.PI * 2 ); ctx.fill(); // Nome abreviado dentro da bolinha. ctx.fillStyle = '#f5ffff'; ctx.textAlign = 'center'; ctx.textBaseline = 'middle'; const lines = mini ? [MINI_DROP_LABELS[d.idx] || '?'] : (ROUND_DROP_LABELS[d.idx] || ['?']); let font = mini ? Math.max(7, Math.min(8.8, radius * .72)) : Math.max(8, Math.min(11, radius * .4)); const maxTextWidth = radius * 1.77; const setFont = () => { ctx.font = `900 ${font}px system-ui`; }; setFont(); while ( lines.some(line => ctx.measureText(line).width > maxTextWidth) && font > 6.5 ) { font -= .25; setFont(); } const lineHeight = font * 1.03; lines.forEach((line, i) => { ctx.fillText( line, cx, cy + (i - (lines.length - 1) / 2) * lineHeight ); }); ctx.restore(); }
